\textbf{Preprint}
\\[2pt]
\textbf{AntiFLipper: A Secure and Efficient Defense Against Label-Flipping Attacks in Federated Learning}\hfill 2025\\[-1pt]
\textbf{A. Rahman}, A. Hasan, S. Arifin, F. H. Bappy, T. Hossain, T. Islam, A. R. M. Kamal, and M. A. Hossain\\[-1pt]
{\small\color{mutedColor}\textit{Under review} \quad arXiv:2509.22873}\\[-1pt]

\quad \href{https://github.com/abidh8820/AntiFlipper}{\faGithub\ Code}\\[-1pt]
Presents a client-level defense against label-flipping attacks that achieves accuracy comparable to existing defenses across IID and non-IID distributions on MNIST and CIFAR-10, with server-side aggregation time comparable to FedAvg. Evaluating on 10\% of local data has an estimated client-side overhead of 1.1\%.

\section{Ongoing Research}
\entry{SPARC-FL: Statistical Peer Auditing with Repeated Confirmation for Free-Rider Detection}{September 2025--Present}{%
    \textit{Master's Thesis $\cdot$ Network and Data Analysis Group (NDAG), Islamic University of Technology}\\[-1pt]
    \href{https://aashnan-rahman.github.io/academic-profile/Resources/Masters/masters_paper.pdf}{\faFile\ Preprint}
    \quad \href{https://github.com/Aashnan-Rahman/Mitigating-Free-Rider}{\faGithub\ Code}\\[-1pt]
    \begin{cvitems}
        \item Designed a server-side framework that audits each client's update norms and layer-wise profiles against its peers, removing a client only after the same evidence recurs across audit cycles, with no auxiliary detector.
        \item Removed all tested free riders across 96 challenge-dispatch runs, with three honest-client removals. In 16 matched runs at 40\% free riders, ordinary dispatch achieved the same removals while withholding fewer honest updates.
    \end{cvitems}
}
